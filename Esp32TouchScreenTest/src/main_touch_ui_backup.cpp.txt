#include <Arduino.h>

#define LGFX_USE_V1
#include <LovyanGFX.hpp>


// ============================================================
// DISPLAY + TOUCH
// ============================================================

class TouchscreenDisplay : public lgfx::LGFX_Device
{
    lgfx::Panel_ILI9341 panel;
    lgfx::Bus_SPI bus;
    lgfx::Light_PWM light;
    lgfx::Touch_XPT2046 touch;

public:
    TouchscreenDisplay()
    {
        // ----------------------------------------------------
        // LCD SPI BUS
        // ----------------------------------------------------
        {
            auto cfg = bus.config();

            cfg.spi_host = VSPI_HOST;
            cfg.spi_mode = 0;

            cfg.freq_write = 40000000;
            cfg.freq_read  = 16000000;

            cfg.spi_3wire = true;
            cfg.use_lock = true;
            cfg.dma_channel = 1;

            cfg.pin_sclk = 14;
            cfg.pin_mosi = 13;
            cfg.pin_miso = -1;
            cfg.pin_dc   = 2;

            bus.config(cfg);
            panel.setBus(&bus);
        }


        // ----------------------------------------------------
        // ILI9341 LCD PANEL
        // ----------------------------------------------------
        {
            auto cfg = panel.config();

            cfg.pin_cs   = 15;
            cfg.pin_rst  = -1;
            cfg.pin_busy = -1;

            cfg.panel_width  = 240;
            cfg.panel_height = 320;

            cfg.offset_x = 0;
            cfg.offset_y = 0;
            cfg.offset_rotation = 0;

            cfg.readable = false;
            cfg.invert = false;
            cfg.rgb_order = false;

            cfg.bus_shared = true;

            panel.config(cfg);
        }


        // ----------------------------------------------------
        // BACKLIGHT
        // ----------------------------------------------------
        {
            auto cfg = light.config();

            cfg.pin_bl = 27;
            cfg.invert = false;
            cfg.freq = 44100;
            cfg.pwm_channel = 7;

            light.config(cfg);
            panel.setLight(&light);
        }


        // ----------------------------------------------------
        // XPT2046 TOUCH CONTROLLER
        // Separate HSPI bus
        // ----------------------------------------------------
        {
            auto cfg = touch.config();

            cfg.x_min = 240;
            cfg.x_max = 3800;
            cfg.y_min = 3700;
            cfg.y_max = 200;

            cfg.pin_int = 36;

            cfg.bus_shared = false;
            cfg.offset_rotation = 0;

            cfg.spi_host = -1;
            cfg.freq = 1000000;

            cfg.pin_sclk = 25;
            cfg.pin_mosi = 32;
            cfg.pin_miso = 39;
            cfg.pin_cs   = 33;

            touch.config(cfg);
            panel.setTouch(&touch);
        }

        setPanel(&panel);
    }

};


TouchscreenDisplay tft;


// ============================================================
// UI LAYOUT
// ============================================================

int screenW = 0;
int screenH = 0;

const int CLEAR_BTN_W = 70;
const int CLEAR_BTN_H = 32;
int clearBtnX = 0;
int clearBtnY = 0;

void drawUI()
{
    tft.fillScreen(TFT_BLACK);

    tft.setTextColor(TFT_CYAN, TFT_BLACK);
    tft.setTextSize(2);
    tft.setCursor(10, 8);
    tft.println("TOUCH TEST");

    // Clear button, top-right corner
    clearBtnX = screenW - CLEAR_BTN_W - 8;
    clearBtnY = 6;
    tft.fillRoundRect(clearBtnX, clearBtnY, CLEAR_BTN_W, CLEAR_BTN_H, 6, TFT_RED);
    tft.setTextColor(TFT_WHITE, TFT_RED);
    tft.setTextSize(1);
    tft.setCursor(clearBtnX + 16, clearBtnY + 12);
    tft.print("CLEAR");

    tft.setTextColor(TFT_WHITE, TFT_BLACK);
    tft.setTextSize(1);
    tft.setCursor(10, 42);
    tft.println("Drag your finger to draw.");
    tft.setCursor(10, 54);
    tft.println("Tap CLEAR to reset.");

    // Draw a border so you can see if touches near the edges
    // register at the edges (a calibration sanity check)
    tft.drawRect(0, 0, screenW, screenH, TFT_DARKGREY);
}

bool touchInClearButton(int x, int y)
{
    return x >= clearBtnX && x <= clearBtnX + CLEAR_BTN_W &&
           y >= clearBtnY && y <= clearBtnY + CLEAR_BTN_H;
}


// ============================================================
// SETUP
// ============================================================

void setup()
{
    Serial.begin(115200);
    delay(500);

    Serial.println();
    Serial.println("LovyanGFX Touch Test starting...");

    tft.init();
    tft.setRotation(1);
    tft.setBrightness(200);

    screenW = tft.width();
    screenH = tft.height();

    drawUI();
}


// ============================================================
// LOOP
// ============================================================

void loop()
{
    int32_t x, y;

    if (tft.getTouch(&x, &y))
    {
        if (touchInClearButton(x, y))
        {
            drawUI();
        }
        else
        {
            tft.fillCircle(x, y, 3, TFT_GREEN);
            Serial.printf("Touch: x=%d y=%d\n", x, y);
        }
    }

    delay(15);
}
